# WS/T 405-2012 临床检验危急值报告与处置规范
# 来源: 国家卫生健康委员会 (2012)
# 状态: 摘要（从临床实践提取核心阈值，原始PDF无法在线获取）

name: 临床检验危急值报告与处置规范
source: 国家卫生健康委员会
version: WS/T 405-2012
trust_level: T1
category: lab_standards

key_content:
  scope: 适用于各级医疗机构临床实验室的危急值报告与管理
  definitions:
    critical_value: 提示患者可能处于生命危险边缘状态的检验结果，需要立即通知临床并采取相应医疗措施
    reporting_chain: 检验科检出 → 电话/信息系统通知临床 → 临床确认 → 处置 → 记录闭环
    turnaround_time: 从确认危急值到通知临床的时间，建议≤15分钟
  core_thresholds:
    potassium: { low: "<2.8 mmol/L", high: ">6.2 mmol/L" }
    sodium: { low: "<120 mmol/L", high: ">160 mmol/L" }
    calcium: { low: "<1.5 mmol/L", high: ">3.5 mmol/L" }
    glucose: { low: "<2.2 mmol/L", high: ">22.2 mmol/L", newborn_low: "<2.0 mmol/L" }
    hemoglobin: { low: "<60 g/L" }
    platelet: { low: "<20×10⁹/L" }
    wbc: { low: "<1.0×10⁹/L", high: ">50×10⁹/L" }
    pt_inr: { high: "INR>5" }
    aptt: { high: ">100秒" }
    ph: { low: "<7.20", high: ">7.60" }
    pCO2: { high: ">70 mmHg" }
    pO2: { low: "<50 mmHg" }
    creatinine: { high: ">530 μmol/L" }
    troponin: { high: "hs-TnI>0.5 ng/mL" }
    bilirubin_newborn: { high: ">342 μmol/L (胆红素脑病危险)" }
    blood_culture: { positive: "立即电话通知菌名+革兰染色" }
    pct: { high: ">10 ng/mL (严重脓毒症)" }
