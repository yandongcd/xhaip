```python
def load_from_dir(dir_path: str, agent_filter: str = "") -> int:
    """加载目录下所有 agent YAML 定义。
    
    Args:
        agent_filter: 非空时仅加载指定 agent + depends_on 链 (BFS).
    """
    if agent_filter:
        return _load_single_agent(dir_path, agent_filter)

    count = 0
    for yf in sorted(Path(dir_path).glob("*.yaml")):
        if _is_skipped(yf.name):
            continue
        try:
            _register_agent(_parse_yaml(yf))
            count += 1
        except Exception as e:
            logger.warning("Agent YAML 加载失败 %s: %s", yf.name, e)
    return count


def _load_single_agent(dir_path: str, agent_name: str) -> int:
    """BFS 加载 agent + depends_on 链."""
    loaded: set[str] = set()
    queue: list[str] = [agent_name]
    count = 0
    dir_p = Path(dir_path)
    while queue:
        name = queue.pop(0)
        if name in loaded:
            continue
        yf = dir_p / f"{name}.yaml"
        if not yf.exists():
            logger.warning("Agent YAML 不存在: %s", yf)
            continue
        agent = _parse_yaml(yf)
        if not agent:
            continue
        _register_agent(agent)
        loaded.add(name)
        count += 1
        for dep in getattr(agent, "depends_on", []):
            dep_name = dep.get("agent", "") or ""
            if dep_name and dep_name not in loaded:
                queue.append(dep_name)
    return count


def _is_skipped(filename: str) -> bool:
    """跳过废弃和内部定义."""
    return ".deprecated" in filename or ".internal" in filename
```
